CREATE VIEW [dbo].[ServicesProjectHours]
AS
     SELECT   h.hs_numberofhours AS Hours,
              CASE
                  WHEN e.hs_rate IS NULL
                  THEN p.hs_projecthourlyrate_base
                  ELSE p.hs_projecthourlyrate * e.hs_rate
              END AS [Hourly Rate],

			  			  CASE 
				  WHEN e.hs_rate IS NULL
				  THEN h.exchangerate
				  ELSE e.hs_rate 
				  END AS [Exchange Rate],

				  CASE WHEN P.hs_projecthourlyrate IS NULL
				  THEN P.hs_projecthourlyrate_base/e.hs_rate
				  ELSE  P.hs_projecthourlyrate END as [Native Hourly Rate],

				  CASE WHEN h.transactioncurrencyidname IS NULL
				  THEN P.transactioncurrencyidname
				  ELSE h.transactioncurrencyidname END  as [Native Currency Type],



							   
              h.hs_numberofhours * (CASE
                                        WHEN e.hs_rate IS NULL
                                        THEN p.hs_projecthourlyrate_base
                                        ELSE p.hs_projecthourlyrate * e.hs_rate
                                    END) AS Revenue,
              CASE
                  WHEN h.hs_billablehours = 1
                       AND p.hs_billableproject = 1
                  THEN 'Yes'
                  ELSE 'No'
              END AS [Billable?],
              a.name AS Account,           
			  convert(date,dbo.fn_ConvertToCentralTime(isnull(h.hs_billeddate, h.hs_dateservicesperformed)))AS [Date],
			  convert(date,dbo.fn_ConvertToCentralTime(h.hs_billeddate)) as BilledDate,
			  dbo.fn_ConvertToCentralTime(h.createdon) as HoursCreated,
			  dbo.fn_ConvertToCentralTime(hs_dateservicesperformed) as DateServicePerformed,
			  CASE
                  WHEN format(h.createdon,'yyyy-MM') > format(h.hs_dateservicesperformed,'yyyy-MM') 
                  THEN 1
                  ELSE 0
              END
			  AS LateEntry,
              p.hs_projectname AS Project,
              h.subject,
              h.ownerid,
			  owneridname.fullname as owneridname,
              p.hs_productbrand AS BrandID,
              hs_projecttaskname.hs_projecttaskname AS ProjectTask,--
              hs_serviceactivitytype_displayname.value as  ServiceType,
              hs_projecttype_displayname.value  AS ProjectType,
              hs_projecthourstype_displayname.value as HoursType,
              CASE
                  WHEN h.hs_forecastedhours = 1
                  THEN 'Yes'
                  ELSE 'No'
              END AS ForecastedHours,
			  p.hs_projectnumber as [Project Number],
              p.hs_projecttype,
              p.hs_projectsid,
              h.hs_forecastedhours,
              p.hs_BudgetHours_Decimal as hs_projectbudgethours,
              c.hs_countryname AS [Account Country],
              hs_projectmanagerprojectleadname.fullname as hs_projectmanagerprojectleadname,
              hs_acenonbillabletype_displayname.value as hs_acenonbillabletype_displayname,
              h.hs_acenonbillabletype,
              p.hs_rsm,
              h.hs_acecomments,
              CASE
                  WHEN p.hs_generatedbyservices = 1
                  THEN 'Yes'
                  ELSE 'No'
              END AS GeneratedByServices,
			  isnull(uc.[Consultant?],'No') as [Consultant?],
			  a.hs_country_picklist1 as BillingCountryID,
			  a.hs_country_picklist2 as ShippingCountryID
     FROM     CRMPROD.dbo.hs_projecthours AS h
	 left join dim_user_consultant uc on h.ownerid = uc.ID
	 left join hswarehouse.dbo.stringmap hs_serviceactivitytype_displayname  on hs_serviceactivitytype = hs_serviceactivitytype_displayname.[attribute value]  and hs_serviceactivitytype_displayname.[object type code] = 'hs_projecthours' and hs_serviceactivitytype_displayname.[attribute name] = 'hs_serviceactivitytype' and hs_serviceactivitytype_displayname.langid = 1033
	 left join hswarehouse.dbo.stringmap hs_acenonbillabletype_displayname  on hs_acenonbillabletype = hs_acenonbillabletype_displayname.[attribute value]  and hs_acenonbillabletype_displayname.[object type code] = 'hs_projecthours' and hs_acenonbillabletype_displayname.[attribute name] = 'hs_acenonbillabletype' and hs_acenonbillabletype_displayname.langid = 1033
              LEFT OUTER JOIN CRMPROD.dbo.hs_projects AS p ON h.hs_project = p.id
			  left join crmprod.dbo.systemuser owneridname on h.ownerid = owneridname.systemuserid
			  left join crmprod.dbo.systemuser hs_projectmanagerprojectleadname on p.hs_projectmanagerprojectlead = hs_projectmanagerprojectleadname.systemuserid
			  left join hswarehouse.dbo.stringmap hs_projecttype_displayname  on hs_projecttype = hs_projecttype_displayname.[attribute value]  and hs_projecttype_displayname.[object type code] = 'hs_projects' and hs_projecttype_displayname.[attribute name] = 'hs_projecttype' and hs_projecttype_displayname.langid = 1033
			  left join hswarehouse.dbo.stringmap hs_projecthourstype_displayname  on hs_projecthourstype = hs_projecthourstype_displayname.[attribute value]  and hs_projecthourstype_displayname.[object type code] = 'hs_projects' and hs_projecthourstype_displayname.[attribute name] = 'hs_projecthourstype' and hs_projecthourstype_displayname.langid = 1033
			  left join crmprod.dbo.hs_projecttasks hs_projecttaskname on h.hs_projecttask = hs_projecttaskname.Id
              LEFT OUTER JOIN CRMPROD.dbo.account AS a ON a.id = p.hs_account
              LEFT OUTER JOIN CRMPROD.dbo.hs_country AS c ON COALESCE(a.hs_country_picklist2,a.hs_country_picklist1) = c.hs_countryoptionsetvalue
              LEFT OUTER JOIN
     (
		 SELECT CONVERT(DATE,hs_enddate) AS EndDate,
       hs_rate,
       hs_currencyone
FROM   CRMPROD.dbo.hs_exchangerates er
       INNER JOIN crmprod.dbo.transactioncurrency tc ON er.hs_currencytwo = tc.transactioncurrencyid
	   where  tc.currencyname = 'US Dollar'
     ) AS e ON e.hs_currencyone = p.transactioncurrencyid
               AND CONVERT(DATE,e.EndDate) = CONVERT(DATE,EOMONTH(h.hs_dateservicesperformed))
WHERE
	 
	 
                  h.hs_dateservicesperformed >=case when datepart(mm,getdate()) <= 3 THEN 
	 dateadd(month,-3,DATEADD(yy, DATEDIFF(yy,0,getdate())-5, 0))
	 ELSE 
	 DATEADD(yy, DATEDIFF(yy,0,getdate())-5, 0)
	 END
	 and 
	 CASE
                  WHEN DATEDIFF(month,h.createdon,h.hs_dateservicesperformed) >= 0
                  THEN h.hs_dateservicesperformed
                  ELSE h.createdon
              END < 
case when datepart(mm,getdate()) >= 10
THEN
dateadd(month,3,DATEADD(yy, DATEDIFF(yy,0,getdate()) + 1, 0))
ELSE
DATEADD(yy, DATEDIFF(yy,0,getdate()) + 1, 0)  End


GO
