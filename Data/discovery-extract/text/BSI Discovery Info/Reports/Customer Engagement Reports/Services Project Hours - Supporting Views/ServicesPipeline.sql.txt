CREATE VIEW [dbo].[ServicesPipeline]
AS
     SELECT hs_productbrand AS [Brand ID], 
            u.fullname AS [RSM], 
            u.systemuserid AS [RSMId], 
            NULL AS [Project Owner], 
            NULL AS [Project Manager], 
            o.name, 
            a.name AS [Account],
            CASE
                WHEN hs_productbrand = '655E48CC-F38D-E911-80FB-0050569F6BB1'
                     AND (hs_totalservices_base = 0
                          OR hs_totalservices_base IS NULL)
                THEN 15000
                ELSE hs_totalservices_base
            END AS hs_totalservices_base, 
            estimatedvalue_base, 
            statuscode_displayname.value AS [Opp Status], 
            hs_confidencepercentage_displayname.value AS hs_confidencepercentage_displayname, 
            CONVERT(DATE, COALESCE(estimatedclosedate, o.createdon)) AS Date, 
            hs_salesstages_displayname.value AS hs_salesstages_displayname, 
            NULL AS [Project Budget Hours], 
            NULL AS [Project Rate], 
            c.hs_countryname AS [Account Country], 
            'Opp' AS [Type]
     FROM crmprod.dbo.opportunity o
          LEFT JOIN hswarehouse.dbo.stringmap statuscode_displayname ON statuscode = statuscode_displayname.[attribute value]
                                                                        AND statuscode_displayname.[object type code] = 'opportunity'
                                                                        AND statuscode_displayname.[attribute name] = 'statuscode'
                                                                        AND statuscode_displayname.langid = 1033
          LEFT JOIN hswarehouse.dbo.stringmap hs_confidencepercentage_displayname ON hs_confidencepercentage = hs_confidencepercentage_displayname.[attribute value]
                                                                                     AND hs_confidencepercentage_displayname.[object type code] = 'opportunity'
                                                                                     AND hs_confidencepercentage_displayname.[attribute name] = 'hs_confidencepercentage'
                                                                                     AND hs_confidencepercentage_displayname.langid = 1033
          LEFT JOIN hswarehouse.dbo.stringmap hs_salesstages_displayname ON hs_salesstages = hs_salesstages_displayname.[attribute value]
                                                                            AND hs_salesstages_displayname.[object type code] = 'opportunity'
                                                                            AND hs_salesstages_displayname.[attribute name] = 'hs_salesstages'
                                                                            AND hs_salesstages_displayname.langid = 1033
          LEFT JOIN crmprod.dbo.systemuser u ON o.ownerid = u.systemuserid
          LEFT JOIN crmprod.dbo.account a ON a.accountid = o.customerid
          LEFT JOIN crmprod.dbo.hs_country c ON COALESCE(a.hs_country_picklist2, hs_country_picklist1) = c.hs_countryoptionsetvalue
     WHERE(o.hs_totalservices_base <> 0
           --AND o.SCRIBE_DELETEDON IS NULL
           AND o.statecode = 0)
          OR (--o.SCRIBE_DELETEDON IS NULL AND 
          o.statecode = 0
          AND o.hs_productbrand IN('524BE826-BF17-E611-80D6-0050569F6BB1', '655E48CC-F38D-E911-80FB-0050569F6BB1'))
     --WHERE        (o.hs_totalservices_base <> 0) /*AND (o.SCRIBE_DELETEDON IS NULL)*/ AND (o.statecode = 0) /*OR
     --                         (o.SCRIBE_DELETEDON IS NULL)*/ AND (o.statecode = 0) AND (o.hs_productbrand IN ('524BE826-BF17-E611-80D6-0050569F6BB1', '655E48CC-F38D-E911-80FB-0050569F6BB1'))
     UNION
     SELECT p.hs_productbrand, 
            rsm.fullname, 
            rsm.systemuserid, 
            po.fullname, 
            pm.fullname, 
            hs_projectname, 
            a.name, 
            hs_BudgetHours_Decimal * hs_projecthourlyrate_base, 
            COALESCE(estimatedvalue_base, hs_BudgetHours_Decimal * hs_projecthourlyrate_base), 
            NULL, 
            NULL, 
            CONVERT(DATE, p.createdon), 
            NULL, 
            hs_BudgetHours_Decimal as hs_projectbudgethours, 
            hs_projecthourlyrate_base, 
            c.hs_countryname AS [Account Country],
            CASE
                WHEN p.statuscode = 717710006
                THEN 'SOW'
                WHEN p.statuscode = 717710004
                THEN 'Pre-Sales'
            END
     FROM crmprod.dbo.hs_projects p
          LEFT JOIN crmprod.dbo.opportunity o ON p.hs_opportunity = o.opportunityid
          LEFT JOIN crmprod.dbo.systemuser rsm ON rsm.systemuserid = p.hs_rsm
          LEFT JOIN crmprod.dbo.systemuser po ON po.systemuserid = p.ownerid
          LEFT JOIN crmprod.dbo.systemuser pm ON pm.systemuserid = p.hs_projectmanagerprojectlead
          LEFT JOIN crmprod.dbo.account a ON p.hs_account = a.accountid
          LEFT JOIN crmprod.dbo.hs_country c ON COALESCE(a.hs_country_picklist2, a.hs_country_picklist1) = c.hs_countryoptionsetvalue
     WHERE p.statuscode IN(717710006, 717710004)
          AND p.statecode = 0
          AND hs_billableproject = 1
          AND (p.hs_opportunity IS NULL
               OR p.hs_opportunity NOT IN
     (
         SELECT opportunityid
         FROM crmprod.dbo.opportunity o
              LEFT JOIN crmprod.dbo.systemuser u ON o.ownerid = u.systemuserid
              LEFT JOIN crmprod.dbo.account a ON a.accountid = o.customerid
              LEFT JOIN crmprod.dbo.hs_country c ON COALESCE(a.hs_country_picklist2, hs_country_picklist1) = c.hs_countryoptionsetvalue
         WHERE(o.hs_totalservices_base <> 0
               --AND o.SCRIBE_DELETEDON IS NULL
               AND o.statecode = 0)
              OR (--o.SCRIBE_DELETEDON IS NULL AND 
              o.statecode = 0
              AND o.hs_productbrand IN('524BE826-BF17-E611-80D6-0050569F6BB1', '655E48CC-F38D-E911-80FB-0050569F6BB1'))
     ));
GO
