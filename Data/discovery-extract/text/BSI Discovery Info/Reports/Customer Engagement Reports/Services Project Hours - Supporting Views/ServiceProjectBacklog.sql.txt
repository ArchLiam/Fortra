CREATE VIEW [dbo].[ServicesProjectBacklog]
AS

SELECT p.hs_projectnumber,
       p.hs_productbrand,
       hs_projecttype_displayname.value AS hs_projecttype_displayname,
       p.hs_projectname,
       p.hs_projectmanagerprojectlead,
       hs_projectmanagerprojectleadname.fullname AS hs_projectmanagerprojectleadname,
       a.name AS Account,
       hs_projecthourstype_displayname.value AS hs_projecthourstype_displayname,
       p.hs_expectedstartdate,
       p.hs_expectedenddate,
       p.hs_projecthourlyrate / r.exchangerate AS HourlyRate,
       p.hs_projecthourlyrate AS HourlyRateNativeCurrency,
	   r.currencyname,
       p.hs_BudgetHours_Decimal - ISNULL(p.hs_BillableHours_Rollup,0) AS hs_projectremaininghours,
       (p.hs_projecthourlyrate / r.exchangerate) * (p.hs_BudgetHours_Decimal - ISNULL(p.hs_BillableHours_Rollup,0)) AS RemainingAmount,
       statuscode_displayname.value AS statuscode_displayname,
       p.ownerid,
       owneridname.fullname AS owneridname,
       CASE
           WHEN p.hs_billableproject = 1
           THEN 'Yes'
           WHEN p.hs_billableproject = 0
           THEN 'No'
       END AS [Billable?],
       CONVERT(DATE,p.createdon) AS createdon,
       p.hs_projecttype,
       p.hs_projectsid,
       CASE
           WHEN DATEADD(month,3,CONVERT(DATE,p.createdon)) > GETDATE()
           THEN '<3 Months'
           WHEN DATEADD(month,6,CONVERT(DATE,p.createdon)) > GETDATE()
           THEN '3-6 Months'
           ELSE '6+ Months'
       END AS [Project Age],
       c.hs_countryname AS [Account Country],
       p.hs_account,
       p.hs_budgetamount_calc_Base as hs_budgetamount_base,
       p.hs_rsm,
       dim.Geths_projectsLink(p.hs_projectsid) AS ProjectLink,
       CASE
           WHEN p.hs_generatedbyservices = 1
           THEN 'Yes'
           ELSE 'No'
       END AS GeneratedByServices,
	   hs_legalentityname,
	   p.hs_firstinvoiceidnumber,
	   ph.exchangerate as FirstExchangeRate
FROM   crmprod.dbo.hs_projects AS p
       LEFT JOIN
(
    SELECT i.invoicenumber,
           le.hs_legalentityname as hs_legalentityname,
           i.exchangerate as exchangerate
    FROM   
		   crmprod.dbo.invoice i
           LEFT JOIN crmprod.dbo.hs_legalentity le ON i.hs_legalentity = le.hs_legalentityid
	WHERE i.hs_project IS NOT NULL
) ph ON p.hs_firstinvoiceidnumber = ph.invoicenumber
       LEFT JOIN [user] hs_projectmanagerprojectleadname ON p.hs_projectmanagerprojectlead = hs_projectmanagerprojectleadname.ID
       LEFT JOIN [user] owneridname ON p.ownerid = owneridname.ID
       LEFT JOIN hswarehouse.dbo.stringmap hs_projecttype_displayname ON p.hs_projecttype = hs_projecttype_displayname.[attribute value]
                                                                         AND hs_projecttype_displayname.[object type code] = 'hs_projects'
                                                                         AND hs_projecttype_displayname.[attribute name] = 'hs_projecttype'
                                                                         AND hs_projecttype_displayname.langid = 1033
       LEFT JOIN hswarehouse.dbo.stringmap hs_projecthourstype_displayname ON p.hs_projecthourstype = hs_projecthourstype_displayname.[attribute value]
                                                                              AND hs_projecthourstype_displayname.[object type code] = 'hs_projects'
                                                                              AND hs_projecthourstype_displayname.[attribute name] = 'hs_projecthourstype'
                                                                              AND hs_projecthourstype_displayname.langid = 1033
       LEFT JOIN hswarehouse.dbo.stringmap statuscode_displayname ON p.statuscode = statuscode_displayname.[attribute value]
                                                                     AND statuscode_displayname.[object type code] = 'hs_projects'
                                                                     AND statuscode_displayname.[attribute name] = 'statuscode'
                                                                     AND statuscode_displayname.langid = 1033
       LEFT OUTER JOIN crmprod.dbo.transactioncurrency AS r ON p.transactioncurrencyid = r.transactioncurrencyid
       LEFT OUTER JOIN crmprod.dbo.account AS a ON a.accountid = p.hs_account
       LEFT OUTER JOIN crmprod.dbo.hs_country AS c ON COALESCE(a.hs_country_picklist2,a.hs_country_picklist1) = c.hs_countryoptionsetvalue
WHERE  (p.statecode = 0)
       --AND (p.statuscode IN(1,717710002,717710003,717710005,717710012,717710000)) --AND (p.SCRIBE_DELETEDON IS NULL)
GO
